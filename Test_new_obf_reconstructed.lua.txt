local PathPartsFolder = Instance.new("Folder")

PathPartsFolder.Parent = Workspace

local function CreatePathPart()
    local part = Instance.new("Part")
    part.Size = Vector3.new(0.6, 0.2, 0.6)
    part.Anchored = true
    part.CanCollide = false
    part.CastShadow = false
    part.Material = Enum.Material.Neon
    part.Color = Color3.fromRGB(0, 255, 255)
    part.Parent = PathPartsFolder
    return part
end

local activePathParts = {}

local function DrawPathParts(waypoints)
    for _, p in ipairs(activePathParts) do
        p:Destroy()
    end
    activePathParts = {}

    for i, wp in ipairs(waypoints) do
        local part = CreatePathPart()
        part.Position = wp.Position + Vector3.new(0, 0.1, 0)
        table.insert(activePathParts, part)
    end
end

local function ClearPathParts()
    for _, p in ipairs(activePathParts) do
        p:Destroy()
    end
    activePathParts = {}
end

-- [[ ฟังก์ชันตรวจสอบว่าอยู่ทีมเดียวกันหรือไม่ ]]
local function isSameTeam(player)
    if not TeamCheckEnabled then return false end
    if player == Client then return true end
    
    if Client.Team and player.Team then
        return Client.Team == player.Team
    end
    
    local success, same = pcall(function()
        if player.TeamColor and Client.TeamColor and player.TeamColor == Client.TeamColor then
            return true
        end
        return false
    end)
    if success and same then return true end
    
    return false
end

-- [[ ฟังก์ชันตรวจสอบว่าเป้าหมายยังปกติและมีชีวิตอยู่หรือไม่ ]]
local function IsTargetValid(player)
    if not player or not player.Parent then return false end
    if player.Character and player.Character:FindFirstChild("Humanoid") then
        local h = player.Character.Humanoid
        if h.Health <= 0 then
            return false
        end
    else
        return false
    end
    return true
end

-- [[ ฟังก์ชันค้นหาผู้เล่นที่ใกล้ที่สุด ]]
local function GetClosestPlayer()
    local closestPlayer = nil
    local shortestDistance = math.huge
    local myRoot = RootPart or (Client.Character and Client.Character:FindFirstChild("HumanoidRootPart"))
    
    if not myRoot then return nil end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= Client and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            if not isSameTeam(player) and not blacklistedPlayers[player] then
                local targetHumanoid = player.Character:FindFirstChild("Humanoid")
                if targetHumanoid and targetHumanoid.Health > 0 then
                    local targetRoot = player.Character.HumanoidRootPart
                    local distance = (myRoot.Position - targetRoot.Position).Magnitude
                    if distance < shortestDistance then
                        shortestDistance = distance
                        closestPlayer = player
                    end
                end
            end
        end
    end

    if not closestPlayer then
        blacklistedPlayers = {}
    end

    return closestPlayer
end

local function GetCurrentMovementPart()
    if Humanoid and Humanoid.SeatPart then
        return Humanoid.SeatPart.AssemblyRootPart or Humanoid.SeatPart
    end
    return RootPart
end

local function ScanForObstacles(currentPart, direction)
    if not currentPart then return false, Vector3.new(0,0,0) end
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterDescendantsInstances = {Character, currentPart.Parent}
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    
    local rayOrigin = currentPart.Position + Vector3.new(0, 1, 0)
    local rayDirection = direction * 12
    local raycastResult = Workspace:Raycast(rayOrigin, rayDirection, raycastParams)
    
    if raycastResult then
        return true, raycastResult.Normal
    end
    return false, Vector3.new(0,0,0)
end

-- ฟังก์ชันหลัก: ระบบวิ่งตามหาเป้าหมาย พร้อมสลับคนใหม่ทันทีถ้าคนเก่าตายหรือไปหาไม่ได้
local function StartRelentlessHoverWalkLoop()
    IsTraveling = true
    local BaseHoverHeight = 4.0
    local CurrentHoverHeight = BaseHoverHeight
    
    local path = PathfindingService:CreatePath({
        AgentRadius = 2.0, 
        AgentHeight = 6.0,
        AgentCanJump = true,
        AgentJumpHeight = 8.0,
        AgentMaxSlope = 89,
        Costs = { 
            BlockedNode = math.huge, 
            DoorArea = 1 
        }
    })

    while IsTraveling and RootPart and Humanoid.Health > 0 do
        local targetPlayer = GetClosestPlayer()
        
        if targetPlayer and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local targetRootPart = targetPlayer.Character.HumanoidRootPart
            local targetPos = targetRootPart.Position
            
            if (RootPart.Position - targetPos).Magnitude > 3.0 then
                local currentStartPos = RootPart.Position
                local success, errorMessage = pcall(function()
                    path:ComputeAsync(currentStartPos, targetPos)
                end)

                if not success or path.Status ~= Enum.PathStatus.Success then
                    blacklistedPlayers[targetPlayer] = true
                    ClearPathParts()
                    task.wait(0.2)
                    continue
                end

                local waypoints = path:GetWaypoints()
                DrawPathParts(waypoints)

                local failedPathCount = 0
                
                for i = 1, #waypoints do
                    if not IsTraveling or Humanoid.Health <= 0 then break end
                    
                    -- ตรวจสอบตลอดเวลาว่าเป้าหมายตายหรือยัง ถ้าตายให้รีบตัดลูปไปหาคนใหม่ทันที
                    if not IsTargetValid(targetPlayer) then
                        blacklistedPlayers[targetPlayer] = true
                        ClearPathParts()
                        break
                    end
                    
                    local wp = waypoints[i]
                    local currentTarget = Vector3.new(wp.Position.X, wp.Position.Y + CurrentHoverHeight, wp.Position.Z)

                    local currentPart = GetCurrentMovementPart()
                    if not currentPart then break end
                    
                    local startPos = currentPart.Position
                    local direction = (currentTarget - startPos).Unit
                    if tostring(direction) == "NAN" then direction = Vector3.new(0,0,0) end
                    
                    local distance = (currentTarget - startPos).Magnitude
                    local movedDistance = 0
                    local startTime = tick()
                    local lastPosCheck = currentPart.Position
                    local stuckTimeCheck = tick()
                    
                    while movedDistance < distance and IsTraveling do
                        local dt = RunService.Heartbeat:Wait()
                        
                        -- เช็คสถานะเป้าหมายระหว่างเดินเข้าหา
                        if not IsTargetValid(targetPlayer) then
                            blacklistedPlayers[targetPlayer] = true
                            ClearPathParts()
                            break
                        end
                        
                        currentPart = GetCurrentMovementPart()
                        if not currentPart or not targetPlayer.Character or not targetPlayer.Character:FindFirstChild("HumanoidRootPart") then break end
                        
                        local currentTargetRoot = targetPlayer.Character.HumanoidRootPart
                        local lookAtPos = Vector3.new(currentTargetRoot.Position.X, currentPart.Position.Y, currentTargetRoot.Position.Z)
                        
                        local isObstacleAhead, avoidanceNormal = ScanForObstacles(currentPart, direction)
                        if isObstacleAhead then
                            direction = (direction + avoidanceNormal * 0.6).Unit
                            CurrentHoverHeight = BaseHoverHeight + 3.0 
                        else
                            CurrentHoverHeight = math.max(BaseHoverHeight, CurrentHoverHeight - (dt * 5))
                        end
                        
                        local elapsedTime = tick() - startTime
                        movedDistance = math.min(elapsedTime * MoveSpeed, distance)
                        local newPos = startPos + (direction * movedDistance)
                        
                        if not Humanoid.SeatPart and Humanoid.Sit then 
                            Humanoid.Sit = false 
                        end
                        
                        if currentPart:IsA("BasePart") then
                            currentPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                            currentPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                        end
                        RootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                        RootPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                        
                        if (lookAtPos - newPos).Magnitude > 0.01 then
                            currentPart:PivotTo(CFrame.lookAt(newPos, lookAtPos))
                        else
                            currentPart:PivotTo(CFrame.new(newPos) * currentPart.CFrame.Rotation)
                        end
                        
                        pcall(function()
                            SendNetwork("set_sprinting_1", true)
                        end)
                        
                        if tick() - stuckTimeCheck >= 0.8 then
                            if (currentPart.Position - lastPosCheck).Magnitude < 1.0 then
                                failedPathCount = failedPathCount + 1
                                if failedPathCount > 2 then
                                    blacklistedPlayers[targetPlayer] = true
                                    break
                                end
                            end
                            lastPosCheck = currentPart.Position
                            stuckTimeCheck = tick()
                        end
                        
                        if Humanoid.Health <= 0 then break end
                    end
                    
                    if not IsTargetValid(targetPlayer) or failedPathCount > 2 then 
                        break 
                    end
                end
            else
                ClearPathParts()
                local currentPart = GetCurrentMovementPart()
                if currentPart and targetRootPart then
                    local lookAtPos = Vector3.new(targetRootPart.Position.X, currentPart.Position.Y, targetRootPart.Position.Z)
                    if (lookAtPos - currentPart.Position).Magnitude > 0.01 then
                        currentPart:PivotTo(CFrame.lookAt(currentPart.Position, lookAtPos))
                    end
                end
                task.wait(0.05)
            end
        else
            ClearPathParts()
            task.wait(0.1)
        end
        
        RunService.Heartbeat:Wait()
    end

    ClearPathParts()
end

-- ฟังก์ชัน Buff ปืน
local function BuffGun(tool)
    if not buffEnabled then return end
    if not tool or not tool:IsA("Tool") then return end

    task.spawn(function()
        task.wait(0.1)
        pcall(function()
            tool:SetAttribute("AutoFire", true)
            tool:SetAttribute("FireRate", 0.001)
            tool:SetAttribute("IsReloading", false)
        end)
    end)
end

-- ฟังก์ชันเชื่อม Backpack + Character ให้ Buff ตลอด
local function SetupConnections()
    -- ยกเลิก Connection เก่า
    for _, conn in ipairs(connections) do
        conn:Disconnect()
    end
    connections = {}

    -- Buff ปืนใน Backpack ตอนนี้
    for _, tool in ipairs(player.Backpack:GetChildren()) do
        BuffGun(tool)
    end

    -- เมื่อมีปืนใหม่เข้ามา Backpack
    table.insert(connections, player.Backpack.ChildAdded:Connect(function(tool)
        BuffGun(tool)
    end))

    -- Buff ปืนที่อยู่ในมือ (Character)
    if player.Character then
        for _, tool in ipairs(player.Character:GetChildren()) do
            BuffGun(tool)
        end

        -- เวลา Equip ของใหม่
        table.insert(connections, player.Character.ChildAdded:Connect(function(tool)
            BuffGun(tool)
        end))
    end
end

-- ต่อใหม่ทุกครั้งที่เกิดใหม่ (Respawn)
table.insert(connections, player.CharacterAdded:Connect(function(char)
    task.wait(0.5)
    SetupConnections()
end))

-- ==================== HELPER FUNCTIONS ====================
local function getTargetPart(character)
    if not character then return nil end
    return character:FindFirstChild("Head") or character:FindFirstChild("UpperTorso")
end

local function isVisible(targetPart)
    if not targetPart then return false end
    if not WallCheckEnabled then return true end
    
    local origin = Camera.CFrame.Position
    local direction = targetPart.Position - origin
    local rayParams = RaycastParams.new()
    
    local ignoreList = {LocalPlayer.Character}
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        table.insert(ignoreList, LocalPlayer.Character)
    end
    if targetPart.Parent then
        table.insert(ignoreList, targetPart.Parent)
    end
    
    rayParams.FilterDescendantsInstances = ignoreList
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    
    local ok, res = pcall(function() 
        return Workspace:Raycast(origin, direction, rayParams) 
    end)
    
    if not ok or not res then return true end
    return res.Instance:IsDescendantOf(targetPart.Parent)
end

local function isSameTeam(player)
    if not TeamCheckEnabled then return false end
    if player == LocalPlayer then return true end
    if LocalPlayer.Team and player.Team then return LocalPlayer.Team == player.Team end
    local success, same = pcall(function()
        if player.TeamColor and LocalPlayer.TeamColor and player.TeamColor == LocalPlayer.TeamColor then
            return true
        end
        return false
    end)
    if success and same then return true end
    return false
end

local function getClosestTarget()
    -- หากเลือกโหมด Sticky แต่ยังไม่ได้กดปุ่มเปิด (Toggle เป็น false) ให้คืนค่า nil
    if LockStyle == "Sticky (keybind)" and not isKeyToggled then
        removeHighlight(lockedTarget)
        lockedTarget = nil
        return nil
    end

    if LockStyle == "Sticky (keybind)" and lockedTarget and lockedTarget.Character then
        local humanoid = lockedTarget.Character:FindFirstChild("Humanoid")
        local targetPart = getTargetPart(lockedTarget.Character)
        if humanoid and humanoid.Health > 0 and targetPart then
            if not WallCheckEnabled or isVisible(targetPart) then
                return lockedTarget
            end
        end
        removeHighlight(lockedTarget)
        lockedTarget = nil
    end

    local closestPlayer = nil
    local shortestDistance = FOV_RADIUS
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local character = player.Character
            if character and character:FindFirstChild("Humanoid") and character.Humanoid.Health > 0 then
                if not isSameTeam(player) then
                    local targetPart = getTargetPart(character)
                    if targetPart then
                        local ok, pos, onScreen = pcall(function() 
                            return Camera:WorldToViewportPoint(targetPart.Position) 
                        end)
                        
                        if ok and onScreen then
                            local distance = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                            if distance < shortestDistance and (not WallCheckEnabled or isVisible(targetPart)) then
                                shortestDistance = distance
                                closestPlayer = player
                            end
                        end
                    end
                end
            end
        end
    end

    if closestPlayer and closestPlayer ~= lockedTarget then
        removeHighlight(lockedTarget)
        lockedTarget = closestPlayer
        if RainbowHighlightEnabled then
            createHighlight(lockedTarget)
        end
    end

    return closestPlayer
end

        if mouse1click then mouse1click()
        elseif mouse1press and mouse1release then mouse1press() task.wait(0.02) mouse1release()
        elseif vim then vim:SendMouseButtonEvent(0, 0, 0, true, game, 0) task.wait(0.02) vim:SendMouseButtonEvent(0, 0, 0, false, game, 0)
        end
    end)
end

RunService.RenderStepped:Connect(function(dt)
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)

    if DrawingIsAvailable and FOV_Circle then
        FOV_Circle.Position = center
        FOV_Circle.Radius = FOV_RADIUS
        FOV_Circle.Visible = AimbotEnabled and ShowFOVEnabled
    end

    colorHue = colorHue + (dt * 0.5)
    if colorHue > 1 then colorHue = 0 end

    if AimbotEnabled then
        local target = getClosestTarget()
        if target and target.Character then
            local targetPart = getTargetPart(target.Character)
            if targetPart then
                if RainbowHighlightEnabled and target.Character:FindFirstChild("AimlockHighlight") then
                    local hl = target.Character:FindFirstChild("AimlockHighlight")
                    hl.FillColor = Color3.fromHSV(colorHue, 1, 1)
                end

                local ok, pos, onScreen = pcall(function()
                    return Camera:WorldToViewportPoint(targetPart.Position)
                end)
                
                if ok then
                    pcall(function()
                        if AimMode == "Hard" then
                            Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, targetPart.Position)
                        else
                            local targetCFrame = CFrame.new(Camera.CFrame.Position, targetPart.Position)
                            Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, Smoothness)
                        end
                    end)
                    
                    if TeleportOrbitEnabled then
                        pcall(function()
                            local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                            if hrp and tick() - lastTweenTime >= tweenSpeed then
                                lastTweenTime = tick()
                                local tPos = targetPart.Position
                                local timeTick = tick() * 6
                                local modeIndex = math.floor(timeTick) % 4
                                local distanceOffset = 4
                                local offsetPos = tPos
                                if modeIndex == 0 then offsetPos = tPos + Vector3.new(0, 0, distanceOffset)
                                elseif modeIndex == 1 then offsetPos = tPos + Vector3.new(0, 0, -distanceOffset)
                                elseif modeIndex == 2 then offsetPos = tPos + Vector3.new(distanceOffset, 0, 0)
                                elseif modeIndex == 3 then offsetPos = tPos + Vector3.new(-distanceOffset, 0, 0)
                                end
                                local targetCF = CFrame.new(offsetPos, Vector3.new(tPos.X, offsetPos.Y, tPos.Z))
                                local tweenInfo = TweenInfo.new(tweenSpeed, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
                                local tween = TweenService:Create(hrp, tweenInfo, {CFrame = targetCF})
                                tween:Play()
                                hrp.AssemblyLinearVelocity = Vector3.zero
                            end
                        end)
                    end
                    
                    if AutoClickEnabled then simulateClick() end
                end
            else
                removeHighlight(lockedTarget)
                lockedTarget = nil
            end
        else
            removeHighlight(lockedTarget)
            lockedTarget = nil
        end
    else
        if lockedTarget then
            removeHighlight(lockedTarget)
            lockedTarget = nil
        end
    end
end)

-- [[ ฟังก์ชันสร้าง ESP ]]
local function CreateESP(Player)
    local Name = Drawing.new("Text")
    
    local Line = Drawing.new("Line")
    local LineOutline = Drawing.new("Line")
    
    local HP_Background = Drawing.new("Line")
    local HP_Bar = Drawing.new("Line")
    
    -- Full Box
    local Box = Drawing.new("Square")
    Box.Visible = false
    Box.Filled = false
    Box.Thickness = 1.5
    Box.Color = Color3.fromRGB(255, 255, 255)

    -- Corners
    local Corners = {}
    local CornerOutlines = {}
    for i = 1, 8 do 
        CornerOutlines[i] = Drawing.new("Line")
        CornerOutlines[i].Thickness = 3.5
        CornerOutlines[i].Color = Color3.fromRGB(0, 0, 0)
        
        Corners[i] = Drawing.new("Line")
        Corners[i].Thickness = 1.5
        Corners[i].Color = Color3.fromRGB(255, 255, 255)
    end

    -- Bone (Skeleton Lines)
    local Bones = {
        -- Head to Spine/UpperTorso
        { "Head", "UpperTorso" }, { "Head", "Head" }, -- R15
        { "Head", "Torso" },                         -- R6
        -- Upper body R15
        { "UpperTorso", "LowerTorso" },
        { "UpperTorso", "LeftUpperArm" }, { "LeftUpperArm", "LeftLowerArm" }, { "LeftLowerArm", "LeftHand" },
        { "UpperTorso", "RightUpperArm" }, { "RightUpperArm", "RightLowerArm" }, { "RightLowerArm", "RightHand" },
        -- Lower body R15
        { "LowerTorso", "LeftUpperLeg" }, { "LeftUpperLeg", "LeftLowerLeg" }, { "LeftLowerLeg", "LeftFoot" },
        { "LowerTorso", "RightUpperLeg" }, { "RightUpperLeg", "RightLowerLeg" }, { "RightLowerLeg", "RightFoot" },
        -- R6 Limbs
        { "Torso", "Left Arm" }, { "Torso", "Right Arm" },
        { "Torso", "Left Leg" }, { "Torso", "Right Leg" }
    }

    local BoneLines = {}
    for _, _ in pairs(Bones) do
        local bLine = Drawing.new("Line")
        bLine.Thickness = 1
        bLine.Color = Color3.fromRGB(255, 255, 255)
        bLine.Visible = false
        table.insert(BoneLines, bLine)
    end

    local function RemoveESP()
        Name:Remove() 
        Line:Remove() LineOutline:Remove() 
        HP_Background:Remove() HP_Bar:Remove()
        Box:Remove()
        for _, v in pairs(Corners) do v:Remove() end
        for _, v in pairs(CornerOutlines) do v:Remove() end
        for _, v in pairs(BoneLines) do v:Remove() end
    end

    RunService.RenderStepped:Connect(function()
        if ESP_Config.Enabled and Player.Character and Player.Character:FindFirstChild("HumanoidRootPart") and Player ~= LocalPlayer and not IsSameTeam(Player) then
            local Root = Player.Character.HumanoidRootPart
            local Hum = Player.Character:FindFirstChild("Humanoid")
            
            local TopPos, OnScreen = Camera:WorldToViewportPoint(Root.Position + Vector3.new(0, 3, 0))
            local BottomPos = Camera:WorldToViewportPoint(Root.Position + Vector3.new(0, -3.5, 0))

            if OnScreen and Hum and Hum.Health > 0 then
                local BoxHeight = math.abs(TopPos.Y - BottomPos.Y)
                local BoxWidth = BoxHeight / 1.8
                local BoxPos = Vector2.new(TopPos.X - (BoxWidth / 2), TopPos.Y)

                -- [Name]
                Name.Visible = ESP_Config.Name
                if Name.Visible then
                    Name.Text = Player.Name
                    Name.Position = Vector2.new(TopPos.X, BoxPos.Y - 22)
                    Name.Outline = true
                    Name.Center = true
                    Name.Size = 15
                    Name.Font = Drawing.Fonts.UI
                    Name.Color = Color3.fromRGB(255, 255, 255)
                end

                -- [Line to Center Top]
                Line.Visible = ESP_Config.Line
                LineOutline.Visible = ESP_Config.Line
                if ESP_Config.Line then
                    LineOutline.Thickness = 3
                    LineOutline.From = Vector2.new(Camera.ViewportSize.X / 2, 0)
                    LineOutline.To = Vector2.new(TopPos.X, TopPos.Y)
                    LineOutline.Color = Color3.fromRGB(0, 0, 0)

                    Line.Thickness = 1
                    Line.From = Vector2.new(Camera.ViewportSize.X / 2, 0)
                    Line.To = Vector2.new(TopPos.X, TopPos.Y)
                    Line.Color = Color3.fromRGB(255, 255, 255)
                end

                -- [Full Box]
                Box.Visible = ESP_Config.FullBox
                if ESP_Config.FullBox then
                    Box.Size = Vector2.new(BoxWidth, BoxHeight)
                    Box.Position = BoxPos
                end

                -- [HP Bar]
                HP_Background.Visible = ESP_Config.HP
                HP_Bar.Visible = ESP_Config.HP
                if ESP_Config.HP then
                    local HP_X = BoxPos.X - 6
                    local currentHeight = BoxHeight * (Hum.Health / Hum.MaxHealth)
                    
                    HP_Background.Thickness = 3
                    HP_Background.From = Vector2.new(HP_X, BoxPos.Y + BoxHeight)
                    HP_Background.To = Vector2.new(HP_X, BoxPos.Y)
                    HP_Background.Color = Color3.fromRGB(0, 0, 0)

                    HP_Bar.Thickness = 1.5
                    HP_Bar.From = Vector2.new(HP_X, BoxPos.Y + BoxHeight)
                    HP_Bar.To = Vector2.new(HP_X, BoxPos.Y + BoxHeight - currentHeight)
                    HP_Bar.Color = Color3.fromRGB(255 - (Hum.Health / Hum.MaxHealth * 255), Hum.Health / Hum.MaxHealth * 255, 0)
                end

                -- [Corner Box]
                local showCorner = ESP_Config.Corner
                for i = 1, 8 do 
                    Corners[i].Visible = showCorner 
                    CornerOutlines[i].Visible = showCorner
                end
                
                if showCorner then
                    local w4 = BoxWidth / 4
                    local h4 = BoxWidth / 4

                    local function setCorner(index, f, t)
                        CornerOutlines[index].From = f
                        CornerOutlines[index].To = t
                        Corners[index].From = f
                        Corners[index].To = t
                    end

                    setCorner(1, BoxPos, BoxPos + Vector2.new(w4, 0))
                    setCorner(2, BoxPos, BoxPos + Vector2.new(0, h4))
                    setCorner(3, Vector2.new(BoxPos.X + BoxWidth, BoxPos.Y), Vector2.new(BoxPos.X + BoxWidth - w4, BoxPos.Y))
                    setCorner(4, Vector2.new(BoxPos.X + BoxWidth, BoxPos.Y), Vector2.new(BoxPos.X + BoxWidth, BoxPos.Y + h4))
                    setCorner(5, Vector2.new(BoxPos.X, BoxPos.Y + BoxHeight), Vector2.new(BoxPos.X + w4, BoxPos.Y + BoxHeight))
                    setCorner(6, Vector2.new(BoxPos.X, BoxPos.Y + BoxHeight), Vector2.new(BoxPos.X, BoxPos.Y + BoxHeight - h4))
                    setCorner(7, Vector2.new(BoxPos.X + BoxWidth, BoxPos.Y + BoxHeight), Vector2.new(BoxPos.X + BoxWidth - w4, BoxPos.Y + BoxHeight))
                    setCorner(8, Vector2.new(BoxPos.X + BoxWidth, BoxPos.Y + BoxHeight), Vector2.new(BoxPos.X + BoxWidth, BoxPos.Y + BoxHeight - h4))
                end

                -- [Bone ESP]
                if ESP_Config.Bone then
                    for index, bonePair in ipairs(Bones) do
                        local boneLine = BoneLines[index]
                        local partA = Player.Character:FindFirstChild(bonePair[1])
                        local partB = Player.Character:FindFirstChild(bonePair[2])

                        if partA and partB then
                            local posA, onScreenA = Camera:WorldToViewportPoint(partA.Position)
                            local posB, onScreenB = Camera:WorldToViewportPoint(partB.Position)

                            if onScreenA and onScreenB then
                                boneLine.From = Vector2.new(posA.X, posA.Y)
                                boneLine.To = Vector2.new(posB.X, posB.Y)
                                boneLine.Visible = true
                            else
                                boneLine.Visible = false
                            end
                        else
                            boneLine.Visible = false
                        end
                    end
                else
                    for _, boneLine in pairs(BoneLines) do
                        boneLine.Visible = false
                    end
                end

            else
                Name.Visible = false 
                Line.Visible = false LineOutline.Visible = false 
                HP_Background.Visible = false HP_Bar.Visible = false
                Box.Visible = false
                for _, v in pairs(Corners) do v.Visible = false end
                for _, v in pairs(CornerOutlines) do v.Visible = false end
                for _, v in pairs(BoneLines) do v.Visible = false end
            end
        else
            Name.Visible = false 
            Line.Visible = false LineOutline.Visible = false 
            HP_Background.Visible = false HP_Bar.Visible = false
            Box.Visible = false
            for _, v in pairs(Corners) do v.Visible = false end
            for _, v in pairs(CornerOutlines) do v.Visible = false end
            for _, v in pairs(BoneLines) do v.Visible = false end
            if not (Player and Player.Parent) then RemoveESP() end
        end
    end)
end

local CFFlyEnabled = false
local FlySpeed = 500

local flyConn

local function ToggleCFFly(state)
    CFFlyEnabled = state

    local char = LocalPlayer.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")

    if flyConn then
        flyConn:Disconnect()
        flyConn = nil
    end

    if not CFFlyEnabled then
        if humanoid then
            humanoid.PlatformStand = false
        end
        return
    end

    flyConn = RunService.RenderStepped:Connect(function(dt)
        local currentCoordChar = LocalPlayer.Character
        local hrp = currentCoordChar and currentCoordChar:FindFirstChild("HumanoidRootPart")
        local hum = currentCoordChar and currentCoordChar:FindFirstChildOfClass("Humanoid")
        local cam = workspace.CurrentCamera
        
        if not hrp or not hum or not cam then return end

        -- เปิด PlatformStand เพื่อไม่ให้ตัวละครล้มหรือโดนแรงโน้มถ่วงดึงตกลงมา
        hum.PlatformStand = true

        local moveDir = Vector3.zero
        if UIS:IsKeyDown(Enum.KeyCode.W) then moveDir += cam.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.S) then moveDir -= cam.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.A) then moveDir -= cam.CFrame.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.D) then moveDir += cam.CFrame.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.Space) then moveDir += Vector3.new(0, 1, 0) end
        if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then moveDir -= Vector3.new(0, 1, 0) end

        if moveDir.Magnitude > 0 then
            -- เคลื่อนที่ไปตามทิศทางที่กด พร้อมหันหน้าตามกล้อง
            local targetPos = hrp.Position + (moveDir.Unit * FlySpeed * dt)
            hrp.CFrame = CFrame.new(targetPos, targetPos + cam.CFrame.LookVector)
        else

-- ค่าเริ่มต้นของแต่ละท่า
local selectedAnims = {
    idle = "Zombie", walk = "Zombie", run = "Zombie",
    jump = "Zombie", fall = "Zombie", climb = "Zombie", sit = "Zombie"
}

local function stopAll()
    for _, track in pairs(currentTracks) do
        pcall(function() track:Stop() end)
    end
    currentTracks = {}

    for _, c in pairs(connections) do
        pcall(function() c:Disconnect() end)
    end
    connections = {}
end

local function applyCustomAnimations(char)
    stopAll()
    if not enabled then return end

    local humanoid = char:WaitForChild("Humanoid")
    local animator = humanoid:FindFirstChildOfClass("Animator") or Instance.new("Animator", humanoid)

    -- ปิด Animate script ดั้งเดิมของ Roblox
    local animateScript = char:FindFirstChild("Animate")
    if animateScript then
        animateScript.Enabled = false
    end

    local tracks = {}
    for animType, packName in pairs(selectedAnims) do
        local packData = ANIM_IDS[packName] or ANIM_IDS["Zombie"]
        local id = packData[animType]
        if id then
            local anim = Instance.new("Animation")
            anim.AnimationId = id
            local track = animator:LoadAnimation(anim)
            
            -- ปรับ Priority ของ Idle ให้เป็น Core เพื่อไม่ให้ไปกวนมือเวลาถือปืน/Tool
            if animType == "idle" then
                track.Priority = Enum.AnimationPriority.Core
            elseif animType == "walk" or animType == "run" then
                track.Priority = Enum.AnimationPriority.Movement
            else
                track.Priority = Enum.AnimationPriority.Action
            end
            
            tracks[animType] = track
        end
    end

    currentTracks = tracks
    local current = nil

    local function play(name)
        if not enabled then return end
        if not tracks[name] then return end
        if current == name then return end
        
        if current and tracks[current] then
            tracks[current]:Stop()
        end
        current = name
        tracks[name]:Play()
    end

    table.insert(connections,
        humanoid.Running:Connect(function(speed)
            if not enabled then return end
            if speed < 1 then
                play("idle")
            else
                play("walk")
            end
        end)
    )

    table.insert(connections,
        humanoid.StateChanged:Connect(function(_, new)
            if not enabled then return end
            if new == Enum.HumanoidStateType.Jumping then
                play("jump")
            elseif new == Enum.HumanoidStateType.Freefall then
                play("fall")
            elseif new == Enum.HumanoidStateType.Climbing then
                play("climb")
            elseif new == Enum.HumanoidStateType.Seated then
                play("sit")
            end
        end)
    )

    play("idle")
end
